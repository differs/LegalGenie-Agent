## 一、需求分析

### 1.1 功能需求

| 需求 | 说明 | 优先级 |
|------|------|--------|
| 记录所有操作 | 增删改查全部记录 | P0 |
| 永久保存 | 不删除，可追溯 | P0 |
| 快速查询 | 按用户/时间/操作类型筛选 | P0 |
| 详细信息 | 记录修改前后的值 | P1 |
| 审计追踪 | 支持操作链还原 | P1 |
| 导出功能 | 支持导出为 Excel/PDF | P2 |

### 1.2 记录范围

| 模块 | 操作类型 | 示例 |
|------|----------|------|
| **用户管理** | 登录、登出、修改密码 | 用户登录系统 |
| **案件管理** | 创建、编辑、删除 | 创建新案件"张三诉李四" |
| **文件管理** | 上传、解析、删除、关联 | 上传证据"合同.pdf" |
| **节点管理** | 创建、编辑、删除、排序 | 创建节点"合同签订" |
| **关联管理** | 绑定、解绑 | 节点关联证据 |
| **系统管理** | 配置修改、权限变更 | 修改用户角色 |

### 1.3 性能要求

| 指标 | 要求 |
|------|------|
| 写入性能 | <50ms（异步写入，不阻塞主流程） |
| 查询性能 | <500ms（100 万条数据内） |
| 存储周期 | 永久保存 |
| 数据量 | 预计 10 万条/年 |

## 二、数据库设计

### 2.1 表结构

```sql
-- 操作日志表
CREATE TABLE operation_logs (
    -- 主键
    id              TEXT PRIMARY KEY,
    
    -- 基础信息
    user_id         TEXT NOT NULL,                    -- 操作人 ID
    user_name       TEXT NOT NULL,                    -- 操作人姓名（冗余，方便查询）
    case_id         TEXT,                             -- 关联案件 ID（可选）
    
    -- 操作信息
    action          TEXT NOT NULL,                    -- 操作类型：CREATE/UPDATE/DELETE/LOGIN/UPLOAD 等
    module          TEXT NOT NULL,                    -- 所属模块：case/file/node/user
    target_type     TEXT NOT NULL,                    -- 目标类型：evidence_file/event_node/case
    target_id       TEXT,                             -- 目标 ID
    target_title    TEXT,                             -- 目标标题（冗余，方便展示）
    
    -- 变更详情
    old_value       JSON,                             -- 修改前的值（UPDATE/DELETE 时记录）
    new_value       JSON,                             -- 修改后的值（CREATE/UPDATE 时记录）
    changed_fields  TEXT,                             -- 变更字段列表：逗号分隔，如 "title,time,description"
    
    -- 环境信息
    ip_address      TEXT,                             -- 操作 IP
    user_agent      TEXT,                             -- 浏览器/客户端信息
    request_id      TEXT,                             -- 请求 ID（用于链路追踪）
    
    -- 时间信息
    created_at      DATETIME DEFAULT CURRENT_TIMESTAMP,  -- 操作时间
    
    -- 索引优化
    INDEX idx_user_time (user_id, created_at),
    INDEX idx_case_time (case_id, created_at),
    INDEX idx_action_time (action, created_at),
    INDEX idx_module_time (module, created_at),
    INDEX idx_target (target_type, target_id)
);
```

### 2.2 索引策略

```sql
-- 单列索引
CREATE INDEX idx_logs_user_id ON operation_logs(user_id);
CREATE INDEX idx_logs_case_id ON operation_logs(case_id);
CREATE INDEX idx_logs_action ON operation_logs(action);
CREATE INDEX idx_logs_module ON operation_logs(module);
CREATE INDEX idx_logs_created_at ON operation_logs(created_at);

-- 复合索引（加速常用查询）
CREATE INDEX idx_logs_user_time ON operation_logs(user_id, created_at DESC);
CREATE INDEX idx_logs_case_time ON operation_logs(case_id, created_at DESC);
CREATE INDEX idx_logs_module_action ON operation_logs(module, action);

-- 全文索引（加速内容搜索）
CREATE VIRTUAL TABLE logs_search USING fts5(
    user_name,
    target_title,
    action,
    module,
    changed_fields,
    content='operation_logs',
    content_rowid='rowid'
);

-- 触发器：自动同步搜索索引
CREATE TRIGGER logs_ai AFTER INSERT ON operation_logs BEGIN
  INSERT INTO logs_search(rowid, user_name, target_title, action, module, changed_fields)
  VALUES (NEW.rowid, NEW.user_name, NEW.target_title, NEW.action, NEW.module, NEW.changed_fields);
END;
```

## 三、Rust 后端实现

### 3.1 数据模型

```rust
// src/models/log.rs
use chrono::{DateTime, Utc};
use serde::{Deserialize, Serialize};
use sqlx::FromRow;
use uuid::Uuid;

/// 操作类型枚举
#[derive(Debug, Clone, Serialize, Deserialize, PartialEq)]
#[serde(rename_all = "SCREAMING_SNAKE_CASE")]
pub enum ActionType {
    Create,
    Update,
    Delete,
    Login,
    Logout,
    Upload,
    Download,
    Parse,
    Link,
    Unlink,
    Export,
}

/// 所属模块
#[derive(Debug, Clone, Serialize, Deserialize, PartialEq)]
#[serde(rename_all = "lowercase")]
pub enum ModuleType {
    Case,
    File,
    Node,
    User,
    System,
}

/// 操作日志结构
#[derive(Debug, Clone, FromRow, Serialize, Deserialize)]
pub struct OperationLog {
    pub id: String,
    pub user_id: String,
    pub user_name: String,
    pub case_id: Option<String>,
    pub action: String,
    pub module: String,
    pub target_type: String,
    pub target_id: Option<String>,
    pub target_title: Option<String>,
    pub old_value: Option<serde_json::Value>,
    pub new_value: Option<serde_json::Value>,
    pub changed_fields: Option<String>,
    pub ip_address: Option<String>,
    pub user_agent: Option<String>,
    pub request_id: Option<String>,
    pub created_at: DateTime<Utc>,
}

impl OperationLog {
    /// 创建新日志
    pub fn new(
        user_id: String,
        user_name: String,
        action: ActionType,
        module: ModuleType,
        target_type: String,
    ) -> Self {
        Self {
            id: Uuid::new_v4().to_string(),
            user_id,
            user_name,
            case_id: None,
            action: format!("{:?}", action),
            module: format!("{:?}", module),
            target_type,
            target_id: None,
            target_title: None,
            old_value: None,
            new_value: None,
            changed_fields: None,
            ip_address: None,
            user_agent: None,
            request_id: None,
            created_at: Utc::now(),
        }
    }
    
    /// 设置变更详情
    pub fn with_changes(
        mut self,
        old_value: Option<serde_json::Value>,
        new_value: Option<serde_json::Value>,
    ) -> Self {
        self.changed_fields = Self::extract_changed_fields(&old_value, &new_value);
        self.old_value = old_value;
        self.new_value = new_value;
        self
    }
    
    /// 提取变更字段列表
    fn extract_changed_fields(
        old_value: &Option<serde_json::Value>,
        new_value: &Option<serde_json::Value>,
    ) -> Option<String> {
        match (old_value, new_value) {
            (Some(Value::Object(old)), Some(Value::Object(new))) => {
                let changed: Vec<String> = old
                    .keys()
                    .filter(|k| old.get(*k) != new.get(*k))
                    .cloned()
                    .collect();
                if changed.is_empty() {
                    None
                } else {
                    Some(changed.join(","))
                }
            }
            _ => None,
        }
    }
}
```

### 3.2 日志服务

```rust
// src/services/log_service.rs
use axum::extract::State;
use sqlx::{SqlitePool, FromRow};
use crate::models::log::OperationLog;
use crate::AppState;

pub struct LogService {
    pool: SqlitePool,
}

impl LogService {
    pub fn new(pool: SqlitePool) -> Self {
        Self { pool }
    }
    
    /// 记录日志（同步写入，简单场景）
    pub async fn log(&self, log: OperationLog) -> anyhow::Result<()> {
        sqlx::query(
            r#"
            INSERT INTO operation_logs (
                id, user_id, user_name, case_id, action, module,
                target_type, target_id, target_title, old_value, new_value,
                changed_fields, ip_address, user_agent, request_id
            ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
            "#,
        )
        .bind(&log.id)
        .bind(&log.user_id)
        .bind(&log.user_name)
        .bind(&log.case_id)
        .bind(&log.action)
        .bind(&log.module)
        .bind(&log.target_type)
        .bind(&log.target_id)
        .bind(&log.target_title)
        .bind(&log.old_value)
        .bind(&log.new_value)
        .bind(&log.changed_fields)
        .bind(&log.ip_address)
        .bind(&log.user_agent)
        .bind(&log.request_id)
        .execute(&self.pool)
        .await?;
        
        Ok(())
    }
    
    /// 查询日志列表
    pub async fn query_logs(
        &self,
        user_id: Option<&str>,
        case_id: Option<&str>,
        module: Option<&str>,
        action: Option<&str>,
        start_time: Option<DateTime<Utc>>,
        end_time: Option<DateTime<Utc>>,
        keyword: Option<&str>,
        page: i64,
        page_size: i64,
    ) -> anyhow::Result<(Vec<OperationLog>, i64)> {
        // 构建动态查询
        let mut where_clauses = Vec::new();
        let mut binds: Vec<Box<dyn sqlx::Encode<sqlx::Sqlite> + Send>> = Vec::new();
        
        if let Some(uid) = user_id {
            where_clauses.push("user_id = ?");
            binds.push(Box::new(uid));
        }
        
        if let Some(cid) = case_id {
            where_clauses.push("case_id = ?");
            binds.push(Box::new(cid));
        }
        
        if let Some(m) = module {
            where_clauses.push("module = ?");
            binds.push(Box::new(m));
        }
        
        if let Some(a) = action {
            where_clauses.push("action = ?");
            binds.push(Box::new(a));
        }
        
        if let Some(start) = start_time {
            where_clauses.push("created_at >= ?");
            binds.push(Box::new(start));
        }
        
        if let Some(end) = end_time {
            where_clauses.push("created_at <= ?");
            binds.push(Box::new(end));
        }
        
        if let Some(kw) = keyword {
            // 使用 FTS5 搜索
            where_clauses.push("target_title LIKE ?");
            binds.push(Box::new(format!("%{}%", kw)));
        }
        
        let where_clause = if where_clauses.is_empty() {
            String::new()
        } else {
            format!("WHERE {}", where_clauses.join(" AND "))
        };
        
        // 查询总数
        let count_query = format!(
            "SELECT COUNT(*) FROM operation_logs {}",
            where_clause
        );
        let total: (i64,) = sqlx::query_as(&count_query)
            .fetch_one(&self.pool)
            .await?;
        
        // 查询数据
        let data_query = format!(
            r#"
            SELECT id, user_id, user_name, case_id, action, module,
                   target_type, target_id, target_title, old_value, new_value,
                   changed_fields, ip_address, user_agent, request_id, created_at
            FROM operation_logs
            {}
            ORDER BY created_at DESC
            LIMIT ? OFFSET ?
            "#,
            where_clause
        );
        
        let logs: Vec<OperationLog> = sqlx::query_as(&data_query)
            .fetch_all(&self.pool)
            .await?;
        
        Ok((logs, total.0))
    }
    
    /// 获取操作历史（针对特定目标）
    pub async fn get_target_history(
        &self,
        target_type: &str,
        target_id: &str,
    ) -> anyhow::Result<Vec<OperationLog>> {
        let logs = sqlx::query_as::<_, OperationLog>(
            r#"
            SELECT * FROM operation_logs
            WHERE target_type = ? AND target_id = ?
            ORDER BY created_at DESC
            LIMIT 100
            "#,
        )
        .bind(target_type)
        .bind(target_id)
        .fetch_all(&self.pool)
        .await?;
        
        Ok(logs)
    }
    
    /// 统计操作次数
    pub async fn count_actions(
        &self,
        user_id: &str,
        start_time: DateTime<Utc>,
        end_time: DateTime<Utc>,
    ) -> anyhow::Result<Vec<(String, i64)>> {
        let stats = sqlx::query_as::<_, (String, i64)>(
            r#"
            SELECT action, COUNT(*) as count
            FROM operation_logs
            WHERE user_id = ? AND created_at BETWEEN ? AND ?
            GROUP BY action
            ORDER BY count DESC
            "#,
        )
        .bind(user_id)
        .bind(start_time)
        .bind(end_time)
        .fetch_all(&self.pool)
        .await?;
        
        Ok(stats)
    }
}
```

### 3.3 日志宏（简化调用）

```rust
// src/macros.rs

/// 记录操作日志宏
#[macro_export]
macro_rules! log_operation {
    (
        $ctx:expr,
        action: $action:expr,
        module: $module:expr,
        target_type: $target_type:expr,
        target_id: $target_id:expr,
        target_title: $target_title:expr,
        old: $old:expr,
        new: $new:expr
    ) => {
        let log = $crate::models::log::OperationLog::new(
            $ctx.user_id.clone(),
            $ctx.user_name.clone(),
            $action,
            $module,
            $target_type.to_string(),
        )
        .with_changes($old, $new);
        
        // 异步写入，不阻塞主流程
        let log_service = $ctx.log_service.clone();
        tokio::spawn(async move {
            if let Err(e) = log_service.log(log).await {
                tracing::error!("Failed to write log: {}", e);
            }
        });
    };
    
    // 简化版（无变更详情）
    (
        $ctx:expr,
        action: $action:expr,
        module: $module:expr,
        target_type: $target_type:expr,
        target_id: $target_id:expr
    ) => {
        log_operation!(
            $ctx,
            action: $action,
            module: $module,
            target_type: $target_type,
            target_id: $target_id,
            target_title: None,
            old: None,
            new: None
        );
    };
}
```

```rust
// frontend/src/pages/logs.rs
use dioxus::prelude::*;
use crate::components::{Table, Pagination, SearchBar};

#[component]
pub fn LogsPage() -> Element {
    let mut filters = use_signal(|| LogFilters::default());
    let logs = use_resource(move || fetch_logs(filters()));
    
    rsx! {
        div { class: "logs-page",
            h1 { "操作日志" }
            
            // 搜索栏
            SearchBar {
                filters: filters(),
                on_search: move |f| filters.set(f),
            }
            
            // 日志列表
            match &*logs.value() {
                Some(Ok(data)) => rsx! {

                        columns: vec![

                            Column { header: "目标", field: "target_title" },
                            Column { header: "变更字段", field: "changed_fields" },
                            Column { header: "详情", field: "actions" },
                        ],

                    }

                        current_page: data.page,
                        total_pages: (data.total / data.page_size) + 1,

                    div { class: "loading", "加载中..." }
                },
            }
        }
    }
```

### 6.1 异步写入

```rust
// 使用消息队列异步写入日志
use tokio::sync::mpsc;

pub struct LogWriter {
    sender: mpsc::Sender<OperationLog>,
}

impl LogWriter {
    pub fn new(pool: SqlitePool) -> Self {
        let (sender, mut receiver) = mpsc::channel(1000);
        
        // 后台任务批量写入
        tokio::spawn(async move {
            let mut buffer = Vec::with_capacity(100);
            
            while let Some(log) = receiver.recv().await {
                buffer.push(log);
                
                // 批量写入（100 条或 1 秒）
                if buffer.len() >= 100 {
                    let _ = bulk_insert_logs(&pool, &buffer).await;
                    buffer.clear();
                }
            }
        });
        
        Self { sender }
    }
    
    pub async fn write(&self, log: OperationLog) {
        let _ = self.sender.send(log).await;
    }
}
```
