| 原则 | 说明 |
|------|------|
| 最小权限 | 用户只拥有完成工作所需的最小权限 |
| 纵深防御 | 多层安全防护，不依赖单一措施 |
| 默认安全 | 默认配置为最安全状态 |
| 故障安全 | 系统故障时进入安全状态 |

```rust
// src/utils/password.rs
use argon2::{
    password_hash::{SaltString, PasswordHash, PasswordHasher, PasswordVerifier},
    Argon2, Algorithm, Params, Version
};
use rand_core::OsRng;

/// 密码强度要求
pub struct PasswordRequirements {
    pub min_length: usize,      // 最小长度 8
    pub require_uppercase: bool, // 大写字母
    pub require_lowercase: bool, // 小写字母
    pub require_digit: bool,     // 数字
    pub require_special: bool,   // 特殊字符
}

impl Default for PasswordRequirements {
    fn default() -> Self {
        Self {
            min_length: 8,
            require_uppercase: true,
            require_lowercase: true,
            require_digit: true,
            require_special: false,
        }
    }
}

/// 验证密码强度
pub fn validate_password_strength(password: &str) -> Result<(), PasswordError> {
    let req = PasswordRequirements::default();
    
    if password.len() < req.min_length {
        return Err(PasswordError::TooShort(req.min_length));
    }
    
    if req.require_uppercase && !password.chars().any(|c| c.is_uppercase()) {
        return Err(PasswordError::MissingUppercase);
    }
    
    if req.require_lowercase && !password.chars().any(|c| c.is_lowercase()) {
        return Err(PasswordError::MissingLowercase);
    }
    
    if req.require_digit && !password.chars().any(|c| c.is_numeric()) {
        return Err(PasswordError::MissingDigit);
    }
    
    // 检查常见弱密码
    let weak_passwords = ["123456", "password", "12345678", "qwerty"];
    if weak_passwords.contains(&password.to_lowercase().as_str()) {
        return Err(PasswordError::TooWeak);
    }
    
    Ok(())
}

/// 哈希密码
pub fn hash_password(password: &str) -> Result<String, anyhow::Error> {
    let salt = SaltString::generate(&mut OsRng);
    
    // Argon2id 参数（OWASP 推荐）
    let params = Params::new(
        19456,  // 内存：19MB
        2,      // 迭代次数
        1,      // 并行度
        Some(Params::DEFAULT_OUTPUT_LEN),
    )?;
    
    let argon2 = Argon2::new(Algorithm::Argon2id, Version::V0x13, params);
    
    let hash = argon2.hash_password(password.as_bytes(), &salt)?;
    Ok(hash.to_string())
}

/// 验证密码
pub fn verify_password(password: &str, hash: &str) -> Result<bool, anyhow::Error> {
    let parsed_hash = PasswordHash::new(hash)?;
    Ok(Argon2::default().verify_password(password.as_bytes(), &parsed_hash).is_ok())
}
```

### 2.2 Token 安全

```rust
// src/utils/token.rs
use jsonwebtoken::{encode, decode, Header, Validation, Algorithm};
use chrono::{Utc, Duration};
use serde::{Deserialize, Serialize};

#[derive(Debug, Serialize, Deserialize)]
pub struct Claims {
    pub sub: String,  // 用户 ID
    pub username: String,
    pub roles: Vec<String>,
    pub exp: usize,   // 过期时间
    pub iat: usize,   // 签发时间
    pub jti: String,  // Token 唯一标识
}

pub fn generate_access_token(user_id: &str, username: &str, roles: &[String]) -> Result<String> {
    let now = Utc::now();
    let exp = now + Duration::minutes(60); // 1 小时有效期
    
    let claims = Claims {
        sub: user_id.to_string(),
        username: username.to_string(),
        roles: roles.to_vec(),
        exp: exp.timestamp() as usize,
        iat: now.timestamp() as usize,
        jti: Uuid::new_v4().to_string(),
    };
    
    let secret = std::env::var("JWT_SECRET")
        .expect("JWT_SECRET must be set");
    
    let token = encode(&Header::default(), &claims, secret.as_bytes())?;
    Ok(token)
}

pub fn verify_token(token: &str) -> Result<Claims> {
    let secret = std::env::var("JWT_SECRET")
        .expect("JWT_SECRET must be set");
    
    // 验证算法
    let mut validation = Validation::new(Algorithm::HS256);
    validation.validate_exp = true;
    validation.leeway = 60; // 60 秒宽容度
    
    let token_data = decode::<Claims>(token, secret.as_bytes(), &validation)?;
    Ok(token_data.claims)
}

// Token 黑名单（用于登出）
pub struct TokenBlacklist {
    blacklist: DashSet<String>,
}

impl TokenBlacklist {
    pub fn add(&self, token: &str, exp: i64) {
        self.blacklist.insert(token.to_string());
        
        // 设置过期清理
        let token = token.to_string();
        tokio::spawn(async move {
            tokio::time::sleep(Duration::from_secs((exp - Utc::now().timestamp()) as u64)).await;
            // 清理黑名单
        });
    }
    
    pub fn is_blacklisted(&self, token: &str) -> bool {
        self.blacklist.contains(token)
    }
}
```

### 2.3 防爆破机制

```rust
// src/middleware/rate_limit.rs
use std::collections::HashMap;
use tokio::sync::Mutex;
use std::time::{Duration, Instant};

pub struct RateLimiter {
    attempts: Mutex<HashMap<String, Vec<Instant>>>,
}

impl RateLimiter {
    pub fn new() -> Self {
        Self {
            attempts: Mutex::new(HashMap::new()),
        }
    }
    
    /// 检查是否超过限制
    pub async fn check_rate_limit(
        &self,
        key: &str,
        max_attempts: usize,
        window: Duration,
    ) -> Result<(), RateLimitError> {
        let mut attempts = self.attempts.lock().await;
        let now = Instant::now();
        
        let user_attempts = attempts.entry(key.to_string()).or_insert_with(Vec::new);
        
        // 清理过期记录
        user_attempts.retain(|&time| now.duration_since(time) < window);
        
        // 检查是否超过限制
        if user_attempts.len() >= max_attempts {
            return Err(RateLimitError::TooManyAttempts);
        }
        
        // 记录本次尝试
        user_attempts.push(now);
        
        Ok(())
    }
}

// 使用示例
#[tokio::test]
async fn test_login_rate_limit() {
    let limiter = RateLimiter::new();
    
    // 前 5 次应该成功
    for _ in 0..5 {
        assert!(limiter.check_rate_limit("user_001", 5, Duration::from_minutes(15)).await.is_ok());
    }
    
    // 第 6 次应该失败
    assert!(limiter.check_rate_limit("user_001", 5, Duration::from_minutes(15)).await.is_err());
}
```

```rust
    }

        return Err(ValidationError::new("password_missing_uppercase"));

/// 文件名验证
pub fn validate_filename(filename: &str) -> Result<(), ValidationError> {
    // 禁止路径遍历
    if filename.contains("..") || filename.contains("/") || filename.contains("\\") {
        return Err(ValidationError::new("invalid_filename"));
    }
    
    // 禁止特殊字符
    let invalid_chars = ['<', '>', ':', '"', '|', '?', '*'];
    if filename.chars().any(|c| invalid_chars.contains(&c)) {
        return Err(ValidationError::new("invalid_filename"));
    }
    
    // 长度限制
    if filename.len() > 255 {
        return Err(ValidationError::new("filename_too_long"));
    }
    
    Ok(())
}
```

### 4.1 文件类型验证

```rust
// src/utils/file_security.rs
use mime_guess::Mime;
use std::path::Path;

/// 允许的文件类型
const ALLOWED_MIME_TYPES: &[&str] = &[
    "application/pdf",
    "application/msword",
    "application/vnd.openxmlformats-officedocument.wordprocessingml.document",
    "application/vnd.ms-excel",
    "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet",
    "image/jpeg",
    "image/png",
    "audio/mpeg",
    "audio/wav",
];

/// 验证文件类型
pub async fn validate_file_type(
    content: &[u8],
    filename: &str,
) -> Result<Mime, FileError> {
    // 1. 检查扩展名
    let ext = Path::new(filename)
        .extension()
        .and_then(|s| s.to_str())
        .ok_or(FileError::InvalidExtension)?;
    
    let mime = mime_guess::from_ext(ext)
        .first()
        .ok_or(FileError::UnknownMimeType)?;
    
    // 2. 检查 MIME 类型
    if !ALLOWED_MIME_TYPES.contains(&mime.as_ref()) {
        return Err(FileError::UnsupportedMimeType);
    }
    
    // 3. 检查文件头（魔术数字）
    let detected_mime = infer::get(content)
        .map(|m| m.mime_type())
        .unwrap_or("application/octet-stream");
    
    // 4. 对比声明和检测的 MIME 类型
    if detected_mime != mime.as_ref() && !is_text_file(detected_mime) {
        return Err(FileError::MimeTypeMismatch);
    }
    
    Ok(mime)
}

/// 检查是否为文本文件（可放宽检查）
fn is_text_file(mime: &str) -> bool {
    mime.starts_with("text/") || mime == "application/json"
}
```

### 4.2 文件上传处理

```rust
// src/routers/files.rs
use axum::extract::Multipart;
use tokio::fs::File;
use tokio::io::AsyncWriteExt;

pub async fn upload_file(
    State(state): State<Arc<AppState>>,
    claims: Claims,
    Path(case_id): Path<String>,
    mut multipart: Multipart,
) -> Result<Json<EvidenceFile>> {
    // 1. 获取文件字段
    let field = multipart.next_field().await?
        .ok_or(FileError::NoFile)?;
    
    let filename = field.file_name()
        .ok_or(FileError::MissingFilename)?
        .to_string();
    
    // 2. 验证文件名
    validate_filename(&filename)?;
    
    // 3. 读取文件内容
    let data = field.bytes().await?;
    
    // 4. 验证文件大小
    if data.len() > state.config.max_file_size {
        return Err(FileError::FileTooLarge(state.config.max_file_size));
    }
    
    // 5. 验证文件类型
    let mime = validate_file_type(&data, &filename).await?;
    
    // 6. 扫描恶意内容（可选）
    if let Some(scanner) = &state.virus_scanner {
        if !scanner.scan(&data).await? {
            return Err(FileError::VirusDetected);
        }
    }
    
    // 7. 生成安全文件名
    let stored_name = generate_safe_filename(&filename);
    let storage_path = format!("files/{}/{}", case_id, stored_name);
    let full_path = state.config.storage_root.join(&storage_path);
    
    // 8. 确保目录存在
    tokio::fs::create_dir_all(full_path.parent().unwrap()).await?;
    
    // 9. 保存文件
    let mut file = File::create(&full_path).await?;
    file.write_all(&data).await?;
    
    // 10. 创建数据库记录
    let evidence = state.db.create_evidence(EvidenceFile {
        id: Uuid::new_v4().to_string(),
        case_id,
        original_name: filename,
        stored_name,
        file_type: mime.to_string(),
        file_size: data.len() as i64,
        storage_path,
        uploaded_by: claims.sub,
        ..Default::default()
    }).await?;
    
    Ok(Json(evidence))
}

/// 生成安全的存储文件名
fn generate_safe_filename(original: &str) -> String {
    let uuid = Uuid::new_v4();
    let ext = Path::new(original)
        .extension()
        .and_then(|s| s.to_str())
        .unwrap_or("bin");
    
    format!("{}.{}", uuid, ext.to_lowercase())
}
```

### 5.1 敏感数据加密

```rust
// src/utils/encryption.rs
use aes_gcm::{
    aead::{Aead, KeyInit, OsRng},
    Aes256Gcm, Nonce
};
use base64::{Engine as _, engine::general_purpose};

/// 加密敏感数据
pub fn encrypt_sensitive_data(data: &str, key: &[u8; 32]) -> Result<String> {
    let cipher = Aes256Gcm::new_from_slice(key)?;
    let nonce = Aes256Gcm::generate_nonce(&mut OsRng);
    
    let ciphertext = cipher.encrypt(&nonce, data.as_bytes())?;
    
    // 拼接 nonce 和密文
    let mut encoded = nonce.to_vec();
    encoded.extend_from_slice(&ciphertext);
    
    Ok(general_purpose::STANDARD.encode(&encoded))
}

/// 解密敏感数据
pub fn decrypt_sensitive_data(encoded: &str, key: &[u8; 32]) -> Result<String> {
    let decoded = general_purpose::STANDARD.decode(encoded)?;
    
    if decoded.len() < 12 {
        return Err(EncryptionError::InvalidData);
    }
    
    let nonce = Nonce::from_slice(&decoded[..12]);
    let ciphertext = &decoded[12..];
    
    let cipher = Aes256Gcm::new_from_slice(key)?;
    let plaintext = cipher.decrypt(nonce, ciphertext)?;
    
    Ok(String::from_utf8(plaintext)?)
}

// 使用示例：加密身份证号
#[tokio::test]
async fn test_encrypt_id_number() {
    let key: [u8; 32] = [0; 32]; // 实际应从环境变量获取
    let id_number = "510101199001011234";
    
    let encrypted = encrypt_sensitive_data(id_number, &key).unwrap();
    let decrypted = decrypt_sensitive_data(&encrypted, &key).unwrap();
    
    assert_eq!(decrypted, id_number);
}
```

### 7.2 HTTP 安全头

```rust
// src/middleware/security_headers.rs
use axum::{
    middleware::Next,
    response::Response,
    http::{Request, header},
};

pub async fn security_headers_middleware<B>(
    mut request: Request<B>,
    next: Next<B>,
) -> Response {
    let mut response = next.run(request).await;
    let headers = response.headers_mut();
    
    // 防止点击劫持
    headers.insert(
        header::X_FRAME_OPTIONS,
        "DENY".parse().unwrap(),
    );
    
    // 防止 MIME 类型嗅探
    headers.insert(
        header::X_CONTENT_TYPE_OPTIONS,
        "nosniff".parse().unwrap(),
    );
    
    // XSS 防护
    headers.insert(
        header::X_XSS_PROTECTION,
        "1; mode=block".parse().unwrap(),
    );
    
    // 内容安全策略
    headers.insert(
        header::CONTENT_SECURITY_POLICY,
        "default-src 'self'; script-src 'self'; style-src 'self' 'unsafe-inline'"
            .parse()
            .unwrap(),
    );
    
    // 严格传输安全
    headers.insert(
        header::STRICT_TRANSPORT_SECURITY,
        "max-age=31536000; includeSubDomains".parse().unwrap(),
    );
    
    // Referrer 策略
    headers.insert(
        header::REFERRER_POLICY,
        "strict-origin-when-cross-origin".parse().unwrap(),
    );
    
    response
}
```
