## 一、测试策略

### 1.1 测试层级

| 层级 | 测试类型 | 工具 | 覆盖率目标 |
|------|----------|------|------------|
| **单元测试** | 函数/方法级别 | cargo test | >80% |
| **集成测试** | 模块间交互 | cargo test + httpx | 核心流程 100% |
| **E2E 测试** | 端到端流程 | Playwright | 关键路径覆盖 |
| **性能测试** | 负载/压力 | wrk / locust | QPS>500 |
| **安全测试** | 漏洞扫描 | cargo-audit | 无高危漏洞 |

### 1.2 测试环境

| 环境 | 用途 | 数据 |
|------|------|------|
| 开发环境 | 开发人员本地测试 | 模拟数据 |
| 测试环境 | CI/CD 自动化测试 | 脱敏数据 |
| 预发布环境 | 上线前验证 | 生产数据副本 |
| 生产环境 | 线上监控 | 真实数据 |

### 1.3 测试流程

```
开发提交代码
    ↓
CI 触发测试
    ↓
┌─────────────────────────────┐
│ 1. 代码检查 (clippy/fmt)     │
│ 2. 单元测试                  │
│ 3. 集成测试                  │
│ 4. E2E 测试 (关键路径)        │
│ 5. 代码覆盖率检查            │
└─────────────────────────────┘
    ↓
全部通过 → 合并代码
    ↓
部署到预发布环境
    ↓
人工验证 → 生产发布
```

```rust
mod tests {

    async fn test_create_case_duplicate_name() {
        let db = setup_test_db().await;
        let service = CaseService::new(db);
        let user_id = "user_test_001";
        
        let input = CreateCaseInput {
            name: "重复案件".to_string(),
            description: None,
            tags: vec![],

        // 创建第一个案件
        service.create_case(user_id, &input).await.unwrap();

            "案件名称已存在"

    }
    
    /// 测试获取案件列表

    async fn test_get_case_list() {
        let db = setup_test_db().await;
        let service = CaseService::new(db);
        let user_id = "user_test_001";
        
        // 创建测试数据
        for i in 0..10 {
            service.create_case(user_id, &CreateCaseInput {
                name: format!("案件{}", i),
                description: None,
                tags: vec![],
            }).await.unwrap();
        }
        
        // 获取列表
        let (cases, total) = service
            .get_case_list(user_id, 1, 20)

            .unwrap();
        
        assert_eq!(total, 10);
        assert_eq!(cases.len(), 10);
    }
```

```rust
// tests/api_tests.rs
use axum::Router;
use http_body_util::BodyExt;
use serde_json::json;
use crate::test_utils::{create_test_app, create_test_user};

/// 测试认证流程
#[tokio::test]
async fn test_auth_flow() {
    let app = create_test_app().await;
    
    // 1. 注册用户
    let register_response = app
        .post("/api/v1/auth/register")
        .json(&json!({
            "username": "testuser",
            "email": "test@example.com",
            "password": "password123"
        }))
        .send()
        .await
        .unwrap();
    
    assert_eq!(register_response.status(), 200);
    
    // 2. 登录获取 Token
    let login_response = app
        .post("/api/v1/auth/login")
        .json(&json!({
            "username": "testuser",
            "password": "password123"
        }))
        .send()
        .await
        .unwrap();
    
    assert_eq!(login_response.status(), 200);
    
    let body: serde_json::Value = login_response.json().await.unwrap();
    let token = body["data"]["access_token"].as_str().unwrap();
    
    // 3. 使用 Token 访问受保护接口
    let me_response = app
        .get("/api/v1/auth/me")
        .header("Authorization", format!("Bearer {}", token))
        .send()
        .await
        .unwrap();
    
    assert_eq!(me_response.status(), 200);
    assert_eq!(me_response["data"]["username"], "testuser");
}

/// 测试案件 CRUD
#[tokio::test]
async fn test_case_crud() {
    let app = create_test_app().await;
    let token = create_test_user(&app).await;
    
    // 1. 创建案件
    let create_response = app
        .post("/api/v1/cases")
        .header("Authorization", format!("Bearer {}", token))
        .json(&json!({
            "name": "测试案件",
            "description": "用于测试的案件"
        }))
        .send()
        .await
        .unwrap();
    
    assert_eq!(create_response.status(), 200);
    let case_id = create_response["data"]["id"].as_str().unwrap();
    
    // 2. 获取案件详情
    let get_response = app
        .get(&format!("/api/v1/cases/{}", case_id))
        .header("Authorization", format!("Bearer {}", token))
        .send()
        .await
        .unwrap();
    
    assert_eq!(get_response.status(), 200);
    assert_eq!(get_response["data"]["name"], "测试案件");
    
    // 3. 更新案件
    let update_response = app
        .put(&format!("/api/v1/cases/{}", case_id))
        .header("Authorization", format!("Bearer {}", token))
        .json(&json!({
            "name": "更新后的案件",
            "description": "更新后的描述"
        }))
        .send()
        .await
        .unwrap();
    
    assert_eq!(update_response.status(), 200);
    
    // 4. 删除案件
    let delete_response = app
        .delete(&format!("/api/v1/cases/{}", case_id))
        .header("Authorization", format!("Bearer {}", token))
        .send()
        .await
        .unwrap();
    
    assert_eq!(delete_response.status(), 200);
    
    // 5. 验证案件已删除
    let get_after_delete = app
        .get(&format!("/api/v1/cases/{}", case_id))
        .header("Authorization", format!("Bearer {}", token))
        .send()
        .await
        .unwrap();
    
    assert_eq!(get_after_delete.status(), 404);
}
```

### 3.2 文件上传测试

```rust
// tests/file_upload_test.rs
use axum::body::Body;
use http::{Request, StatusCode};
use mime_guess::mime;

#[tokio::test]
async fn test_file_upload() {
    let app = create_test_app().await;
    let token = create_test_user(&app).await;
    
    // 创建测试案件
    let case_id = create_test_case(&app, &token).await;
    
    // 准备测试文件
    let file_content = b"测试文件内容";
    let boundary = "----WebKitFormBoundary";
    
    let body = format!(
        "--{}\r\n\
         Content-Disposition: form-data; name=\"file\"; filename=\"test.pdf\"\r\n\
         Content-Type: application/pdf\r\n\r\n\
         {}\r\n\
         --{}--\r\n",
        boundary,
        String::from_utf8_lossy(file_content),
        boundary
    );
    
    // 上传文件
    let upload_response = app
        .post(&format!("/api/v1/cases/{}/files", case_id))
        .header("Authorization", format!("Bearer {}", token))
        .header("Content-Type", format!("multipart/form-data; boundary={}", boundary))
        .body(body)
        .send()
        .await
        .unwrap();
    
    assert_eq!(upload_response.status(), 200);
    
    let evidence_id = upload_response["data"]["id"].as_str().unwrap();
    assert!(evidence_id.starts_with("evidence_"));
}

#[tokio::test]
async fn test_file_upload_too_large() {
    let app = create_test_app().await;
    let token = create_test_user(&app).await;
    let case_id = create_test_case(&app, &token).await;
    
    // 准备超大文件 (超过 100MB)
    let large_content = vec![0u8; 101 * 1024 * 1024];
    
    let upload_response = app
        .post(&format!("/api/v1/cases/{}/files", case_id))
        .header("Authorization", format!("Bearer {}", token))
        .body(large_content)
        .send()
        .await
        .unwrap();
    
    assert_eq!(upload_response.status(), StatusCode::BAD_REQUEST);
    assert_eq!(upload_response["code"], 420102);
}
```

```python
# tests/performance/locustfile.py
from locust import HttpUser, task, between
import random

class LegalMindsUser(HttpUser):
    wait_time = between(1, 3)
    
    def on_start(self):
        # 登录获取 Token
        response = self.client.post("/api/v1/auth/login", json={
            "username": "testuser",
            "password": "password123"
        })
        self.token = response.json()["data"]["access_token"]
        self.headers = {"Authorization": f"Bearer {self.token}"}
    
    @task(3)
    def get_cases(self):
        self.client.get("/api/v1/cases", headers=self.headers)
    
    @task(2)
    def get_timeline(self):
        case_id = random.choice(["case_001", "case_002", "case_003"])
        self.client.get(
            f"/api/v1/cases/{case_id}/timeline/nodes",
            headers=self.headers
        )
    
    @task(1)
    def search(self):
        keywords = ["合同", "证据", "张三", "李四"]
        keyword = random.choice(keywords)
        self.client.get(
            f"/api/v1/search?keyword={keyword}",
            headers=self.headers
        )
```
