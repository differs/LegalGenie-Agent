### 3.3 许可证生成（服务器端）

```rust
use ring::{signature, rand};
use base64::{Engine as _, engine::general_purpose};

const PRIVATE_KEY_PKCS8: &[u8] = include_bytes!("private_key.pkcs8");

pub fn generate_license(
    license_type: &str,
    user_email: &str,
    days: i64,
    machine_id: &str,
) -> Result<LicenseInfo> {
    let now = chrono::Utc::now();
    
    // 准备数据
    let data = format!(
        "{}:{}:{}:{}:{}",
        license_type,
        user_email,
        now.timestamp(),
        (now + chrono::Duration::days(days)).timestamp(),
        machine_id,
    );
    
    // 私钥签名
    let key_pair = signature::Ed25519KeyPair::from_pkcs8(PRIVATE_KEY_PKCS8)?;
    let sig = key_pair.sign(data.as_bytes());
    
    Ok(LicenseInfo {
        license_type: license_type.to_string(),
        user_email: user_email.to_string(),
        issued_at: now,
        expires_at: Some(now + chrono::Duration::days(days)),
        machine_id: machine_id.to_string(),
        signature: general_purpose::STANDARD.encode(sig.as_ref()),
    })
}
```

### 3.4 许可证验证（客户端）

```rust
use ring::signature;
use base64::{Engine as _, engine::general_purpose};

const PUBLIC_KEY_PEM: &str = r#"
-----BEGIN PUBLIC KEY-----
MCowBQYDK2VwAyEA...（公钥）
-----END PUBLIC KEY-----
"#;

pub fn verify_license(license: &LicenseInfo) -> bool {
    // 1. 重建数据
    let data = format!(
        "{}:{}:{}:{}:{}",
        license.license_type,
        license.user_email,
        license.issued_at.timestamp(),
        license.expires_at.map(|e| e.timestamp()).unwrap_or(0),
        license.machine_id,
    );
    
    // 2. 解码签名
    let sig = match general_purpose::STANDARD.decode(&license.signature) {
        Ok(s) => s,
        Err(_) => return false,
    };
    
    // 3. 公钥验证
    let public_key = parse_pem_public_key(PUBLIC_KEY_PEM).unwrap();
    let unparsed_pk = signature::UnparsedPublicKey::new(
        &signature::ED25519,
        public_key,
    );
    
    unparsed_pk.verify(data.as_bytes(), &sig).is_ok()
}
```

```rust
// src/utils/machine_id.rs
use machine_uid::get_machine_uid;
use sha2::{Sha256, Digest};

pub fn get_machine_id() -> Result<String> {
    let raw_id = get_machine_uid()?;
    
    let mut hasher = Sha256::new();
    hasher.update(raw_id.as_bytes());
    let hash = hasher.finalize();
    
    Ok(format!("{:x}", hash))
}
```

```rust
pub fn verify_machine_binding(license: &LicenseInfo) -> bool {

    license.machine_id == current_machine_id
```

## 五、许可证管理器

### 5.1 本地存储

```
~/.config/legalminds/
└── license.json
```

### 5.2 管理器实现

```rust
// src/license/manager.rs
use std::path::PathBuf;

pub struct LicenseManager {
    license: Option<LicenseInfo>,
    license_path: PathBuf,
}

impl LicenseManager {
    pub fn new() -> Result<Self> {
        let license_path = dirs::config_dir()
            .map(|d| d.join("legalminds").join("license.json"))
            .ok_or_else(|| anyhow!("无法找到配置目录"))?;
        
        Ok(Self { license: None, license_path })
    }
    
    /// 加载许可证
    pub fn load_license(&mut self) -> Result<()> {
        if self.license_path.exists() {
            let content = std::fs::read_to_string(&self.license_path)?;
            let license: LicenseInfo = serde_json::from_str(&content)?;
            
            if verify_license(&license) {
                self.license = Some(license);
                Ok(())
            } else {
                Err(anyhow!("许可证验证失败"))
            }
        } else {
            Err(anyhow!("未找到许可证文件"))
        }
    }
    
    /// 激活许可证
    pub fn activate(&mut self, license_key: &str, email: &str) -> Result<LicenseInfo> {
        let machine_id = get_machine_id()?;
        
        // 调用服务器激活
        let license = activate_online(license_key, email, &machine_id).await?;
        
        // 验证
        if !verify_license(&license) {
            return Err(anyhow!("许可证验证失败"));
        }
        
        // 保存
        self.save_license(&license)?;
        self.license = Some(license.clone());
        
        Ok(license)
    }
    
    /// 检查状态
    pub fn check_status(&self) -> LicenseStatus {
        match &self.license {
            None => LicenseStatus::NotFound,
            Some(license) => {
                if !verify_license(license) {
                    LicenseStatus::Invalid
                } else if license.is_expired() {
                    LicenseStatus::Expired
                } else if !verify_machine_binding(license) {
                    LicenseStatus::MachineMismatch
                } else {
                    LicenseStatus::Valid(license.clone())
                }
            }
        }
    }
    
    /// 获取水印文本
    pub fn get_watermark_text(&self) -> String {
        match &self.license {
            Some(license) if license.is_trial() => {
                format!("🔒 测试版 - {} - 有效期至：{}",
                    license.user_email,
                    license.expires_at
                        .map(|e| e.format("%Y-%m-%d").to_string())
                        .unwrap_or_else(|| "未知".to_string())
                )
            }
            _ => String::new(),
        }
    }
}
```

## 六、激活界面

### 6.1 前端组件

```rust
// frontend/src/license.rs
#[component]
pub fn LicenseModal(on_close: EventHandler<()>) -> Element {
    let mut license_key = use_signal(|| String::new());
    let mut email = use_signal(|| String::new());
    let mut error = use_signal(|| Option::<String>::None);
    
    let handle_activate = move |_| {
        let key = license_key();
        let mail = email();
        
        spawn(async move {
            match activate_license(&key, &mail).await {
                Ok(_) => on_close.call(()),
                Err(e) => error.set(Some(e.to_string())),
            }
        });
    };
    
    rsx! {
        div { class: "modal-overlay",
            div { class: "modal license-modal",
                h2 { "激活许可证" }
                
                if let Some(err) = error() {
                    div { class: "error-message", {err} }
                }
                
                input {
                    type: "email",
                    placeholder: "注册邮箱",
                    value: email(),
                    oninput: move |e| email.set(e.value()),
                }
                
                input {
                    type: "text",
                    placeholder: "许可证密钥",
                    value: license_key(),
                    oninput: move |e| license_key.set(e.value()),
                }
                
                button {
                    class: "primary",
                    onclick: handle_activate,
                    "激活"
                }
                
                div { class: "trial-info",
                    h4 { "📋 测试版说明" }
                    ul {
                        li { "测试有效期：30 天" }
                        li { "功能限制：部分高级功能不可用" }
                        li { "导出文件带有测试版水印" }
                        li { "仅限个人评估使用" }
                    }
                }
            }
        }
    }
}
```

## 十、常见问题

### Q1: 水印能去掉吗？
> 技术上能，但需要修改代码。有许可证验证的话，改了也打不开软件。

### Q2: 许可证能复制吗？
> 不能。绑定了机器码，复制到其他电脑验证不通过。

### Q3: 离线能用吗？
> 可以。测试版 30 天内无需联网。正式版可设置 7 天离线宽限期。

### Q4: 被破解了怎么办？
> 1. 在线验证可远程吊销
> 2. 下次更新加强验证
> 3. 法律手段（如有商业损失）
